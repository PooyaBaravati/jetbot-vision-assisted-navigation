cmake_minimum_required(VERSION 3.0.2)
project(cl_camera_plugins)
add_compile_options(-std=c++11 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp costmap_2d nav_core pluginlib tf2_ros tf2_geometry_msgs sensor_msgs std_msgs geometry_msgs nav_msgs map_msgs actionlib_msgs move_base_msgs)
catkin_package(INCLUDE_DIRS include LIBRARIES cl_camera_plugins CATKIN_DEPENDS roscpp costmap_2d nav_core pluginlib tf2_ros tf2_geometry_msgs sensor_msgs std_msgs geometry_msgs nav_msgs map_msgs actionlib_msgs move_base_msgs)
include_directories(include ${catkin_INCLUDE_DIRS})
add_library(cl_camera_plugins src/expiring_camera_layer.cpp src/camera_turn_recovery.cpp)
add_dependencies(cl_camera_plugins ${catkin_EXPORTED_TARGETS})
target_link_libraries(cl_camera_plugins ${catkin_LIBRARIES})
install(TARGETS cl_camera_plugins LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(FILES costmap_plugins.xml recovery_plugins.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
